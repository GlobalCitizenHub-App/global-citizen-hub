const continentList = ["Africa", "Asia", "Europe", "North America", "South America", "Antarctica", "Oceania"]; // Tells the user to pick a city if they type a country const countryCityMap = { "china": "Shanghai", "new zealand": "Auckland", "japan": "Tokyo", "south korea": "Seoul", "korea": "Seoul", "turkey": "Istanbul", "france": "Paris", "united kingdom": "London", "uk": "London", "australia": "Sydney", "canada": "Vancouver" }; // Disambiguates tricky US towns const ambiguousTownsMap = { "aurora": "Aurora, CO or Aurora, IL", "middletown": "Middletown, DE or Middletown, NY", "springfield": "Springfield, IL or Springfield, MA" }; // NEW: Automatically completes known global hubs with their correct country const smartCityComplete = { "seoul": "Seoul, South Korea", "tokyo": "Tokyo, Japan", "paris": "Paris, France", "london": "London, UK", "sydney": "Sydney, Australia", "vancouver": "Vancouver, Canada", "istanbul": "Istanbul, Turkey", "dubai": "Dubai, UAE", "singapore": "Singapore" }; function checkMainFeedback(inputId, feedbackId) { const inputElem = document.getElementById(inputId); let val = inputElem.value.trim(); const feedbackEl = document.getElementById(feedbackId); if (!val) { feedbackEl.innerText = "⚠️ Please enter a specific destination city"; feedbackEl.className = "feedback-tag feedback-warning"; updateAllLinks(); return; } let valLower = val.toLowerCase(); // Auto-normalize US/UK abbreviations if (valLower === "us" || valLower === "usa") { val = "US"; inputElem.value = val; valLower = "us"; } else if (valLower === "uk") { val = "UK"; inputElem.value = val; valLower = "uk"; } // NEW: Smart Auto-Complete for Global Hubs if (smartCityComplete[valLower]) { val = smartCityComplete[valLower]; inputElem.value = val; // Updates the input box automatically valLower = val.toLowerCase(); } const isContinent = continentList.some(c => c.toLowerCase() === valLower); const specificCityExample = countryCityMap[valLower]; const specificSuggestion = ambiguousTownsMap[valLower]; if (isContinent) { feedbackEl.innerText = `⚠️ ${val} is a continent, please enter a specific city name.`; feedbackEl.className = "feedback-tag feedback-warning"; } else if (specificCityExample) { feedbackEl.innerText = `⚠️ ${val} is a country, please enter a specific city (e.g., ${specificCityExample}).`; feedbackEl.className = "feedback-tag feedback-warning"; } else if (specificSuggestion) { feedbackEl.innerText = `⚠️ Please specify state (e.g., ${specificSuggestion}).`; feedbackEl.className = "feedback-tag feedback-warning"; } else { feedbackEl.innerText = "✓ Confirmed destination scope active"; feedbackEl.className = "feedback-tag feedback-valid"; } updateAllLinks(); }
